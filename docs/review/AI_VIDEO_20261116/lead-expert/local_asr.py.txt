#!/usr/bin/env python3
"""Local cached faster-whisper only. Never download a model or call a cloud API."""
import argparse, hashlib, json, re, sys
from pathlib import Path


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('--file', required=True, type=Path)
    p.add_argument('--model', required=True, type=Path)
    p.add_argument('--language', default='zh')
    p.add_argument('--expected')
    args = p.parse_args()
    if not args.file.is_file() or not args.model.is_dir():
        p.error('input media and an existing local model directory are required; no automatic download')
    try:
        from faster_whisper import WhisperModel
    except ImportError:
        p.error('faster_whisper is unavailable in this Python; use the existing project ASR environment with --asr-python')
    h = hashlib.sha256()
    with args.file.open('rb') as f:
        for block in iter(lambda: f.read(8 * 1024 * 1024), b''):
            h.update(block)
    model = WhisperModel(str(args.model.resolve()), device='cpu', compute_type='int8', local_files_only=True)
    segments, info = model.transcribe(str(args.file.resolve()), language=args.language, beam_size=5, word_timestamps=True)
    rows = []
    for segment in segments:
        rows.append({'start': segment.start, 'end': segment.end, 'text': segment.text,
                     'words': [{'start': w.start, 'end': w.end, 'word': w.word, 'probability': w.probability} for w in (segment.words or [])]})
    text = ''.join(s['text'] for s in rows)
    norm = lambda s: re.sub(r'[^\w\u4e00-\u9fff]', '', s)
    print(json.dumps({'schema': 'dsh.ai-video-local-asr/1', 'source_sha256': h.hexdigest(),
                      'model': str(args.model.resolve()), 'language': info.language, 'duration': info.duration,
                      'status': 'review_required', 'text': text, 'segments': rows,
                      'expected': args.expected,
                      'normalized_text_match': None if args.expected is None else norm(text) == norm(args.expected),
                      'limits': ['ASR may substitute homophones or omit words; review against approved dialogue.',
                                 'Word timestamps are estimates, not frame-accurate speech boundaries.',
                                 'This report does not prove lip sync, timbre, acting, or listener approval.']}, ensure_ascii=False))


if __name__ == '__main__':
    main()
